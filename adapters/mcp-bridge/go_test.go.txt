// Only compiled into an Inspector-owned temporary test snapshot. No public API.
func TestInspectorMCPBridge(t *testing.T) {
	dir := os.Getenv("SAGE_BRIDGE_DIR")
	role := os.Getenv("SAGE_BRIDGE_ROLE")
	if dir == "" || (role != "client" && role != "server") {
		t.Fatal("missing bridge configuration")
	}
	fixture, host := hostFixture(t)
	protected := os.Getenv("SAGE_BRIDGE_PROTECTED") == "1"
	alice, bob, endpointClock := setupEndpoints(t)
	cfg := mcpConnectionConfig{endpoint: bob, name: "inspector", version: "1", ttl: 300, timeout: 10 * time.Second}
	var conn net.Conn
	var err error
	if role == "server" {
		listener, e := net.Listen("tcp", "127.0.0.1:0")
		if e != nil {
			t.Fatal(e)
		}
		defer listener.Close()
		if e = listener.(*net.TCPListener).SetDeadline(time.Now().Add(10 * time.Second)); e != nil {
			t.Fatal(e)
		}
		if e = bridgeWrite(filepath.Join(dir, "address"), []byte(listener.Addr().String())); e != nil {
			t.Fatal(e)
		}
		conn, err = listener.Accept()
	} else {
		addr := os.Getenv("SAGE_BRIDGE_PEER")
		endpoint, e := net.ResolveTCPAddr("tcp", addr)
		if e != nil || endpoint.IP.String() != "127.0.0.1" {
			t.Fatal("loopback required")
		}
		conn, err = net.DialTimeout("tcp", addr, 3*time.Second)
		cfg.endpoint = alice
		cfg.initiator = true
		cfg.recipient = setupBob
		cfg.recipientKey = setupBob + "#signing-1"
	}
	if err != nil {
		t.Fatal(err)
	}
	ready := false
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	err = host.connection(ctx, conn, cfg, func(context.Context) error { return nil }, func(ctx context.Context, s *mcpSetupSession, stream mcpSetupIO) error {
		s.owner.mu.Lock()
		live := s.owner.phase == mcpReady
		s.owner.mu.Unlock()
		if !live {
			return ErrInvalid
		}
		if protected {
			intent, e := os.ReadFile(filepath.Join(dir, "intent.json"))
			if e != nil {
				return e
			}
			if !bytes.Equal(intent, fixture.intent) {
				t.Error("common intent differs from core fixture")
				return ErrInvalid
			}
			fixture.executor.output = []byte(`{"text":"inert public fixture"}`)
			if role == "client" {
				client, e := openMCPOwnedClient(ctx, s, host.clients, filepath.Join(dir, "client.journal"), true, intent, fixture.config.authority, fixture.config.resultAuthority, fixture.config.policy, replyClientClock{fixture})
				if e != nil {
					return e
				}
				defer client.client.Close()
				var delivery *ClientDelivery
				attempts := 0
				for attempts < 4 {
					if attempts > 0 {
						if e := bridgeWrite(filepath.Join(dir, "clock"), []byte(strconv.Itoa(360000+attempts*1000))); e != nil {
							return e
						}
					}
					bridgeSyncClock(t, dir, fixture.clock, endpointClock)
					attempts++
					delivery, e = client.exchange(ctx, stream)
					if e != nil {
						return e
					}
					if delivery.Status() == "completed" {
						break
					}
					if delivery.Status() != "pending" {
						return ErrInvalid
					}
					time.Sleep(10 * time.Millisecond)
				}
				if delivery == nil || !delivery.FirstTerminal() || string(delivery.Output()) != `{"text":"inert public fixture"}` {
					return ErrInvalid
				}
				if _, e = client.exchange(ctx, stream); e == nil {
					return ErrInvalid
				}
				raw, _ := json.Marshal(map[string]any{"role": role, "state": "READY", "status": "completed", "first_terminal": true, "output_hex": fmtHex(delivery.Output()), "attempts": attempts, "repeat_denied": true})
				if e = bridgeWrite(filepath.Join(dir, role+".json"), raw); e != nil {
					return e
				}
			} else {
				for {
					wire, e := stream.Receive(ctx)
					if e != nil {
						break
					}
					bridgeSyncClock(t, dir, fixture.clock, endpointClock)
					if _, e = s.admitProtected(ctx, wire, host.gate); e != nil {
						return e
					}
					if e = s.replyProtected(ctx, stream); e != nil {
						return e
					}
				}
				if fixture.executor.effects.Load() != 1 {
					return ErrInvalid
				}
				raw, _ := json.Marshal(map[string]any{"role": role, "state": "READY", "effects": fixture.executor.effects.Load()})
				if e := bridgeWrite(filepath.Join(dir, role+".json"), raw); e != nil {
					return e
				}
			}
			ready = true
			return nil
		}
		raw, _ := json.Marshal(map[string]any{"role": role, "state": "READY", "protected": "NOT_RUN"})
		if e := bridgeWrite(filepath.Join(dir, role+".json"), raw); e != nil {
			return e
		}
		ready = true
		end := time.Now().Add(10 * time.Second)
		for time.Now().Before(end) {
			if _, e := os.Stat(filepath.Join(dir, "release")); e == nil {
				return nil
			}
			time.Sleep(time.Millisecond)
		}
		return ErrInvalid
	})
	// Peer shutdown after the shared release may race the final liveness check.
	// Only READY before release is evidence; post-release delivery is not claimed.
	if !ready {
		t.Fatal("setup did not reach READY", err)
	}
	if _, e := os.Stat(filepath.Join(dir, "release")); !protected && e != nil {
		t.Fatal("controller did not release")
	}
	stop, cancelStop := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancelStop()
	if e := host.stop(stop); e != nil {
		t.Fatal(e)
	}
	if protected && role == "server" {
		if e := fixture.gate.close(); e != nil {
			t.Fatal(e)
		}
		raw, e := os.ReadFile(fixture.path)
		if e != nil {
			t.Fatal(e)
		}
		if e = bridgeWrite(filepath.Join(dir, "server.journal"), raw); e != nil {
			t.Fatal(e)
		}
	}
}

func bridgeWrite(path string, raw []byte) error {
	if err := os.WriteFile(path+".tmp", raw, 0600); err != nil {
		return err
	}
	return os.Rename(path+".tmp", path)
}

func fmtHex(raw []byte) string { return hex.EncodeToString(raw) }

func bridgeSyncClock(t *testing.T, dir string, clocks ...*setupRegistry) {
	raw, e := os.ReadFile(filepath.Join(dir, "clock"))
	if e != nil {
		t.Fatal(e)
	}
	value, e := strconv.ParseInt(string(raw), 10, 64)
	if e != nil {
		t.Fatal(e)
	}
	for _, c := range clocks {
		c.mono.Store(value)
	}
}
